cmake_minimum_required(VERSION 3.24)
project(ZCodePet VERSION 3.0.0 LANGUAGES C CXX)
enable_language(RC)
configure_file(src/resources.rc.in ${CMAKE_CURRENT_BINARY_DIR}/resources.rc @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(ZCodePet WIN32 src/main.cpp src/data_sources.cpp src/zcode_source.cpp src/deepseek_source.cpp src/harness_cache.cpp src/toolbar.cpp src/status_badge.cpp src/notice_close.cpp src/popup_motion.cpp src/frame_clock.cpp src/session_navigation.cpp src/interaction_motion.cpp src/panel_composition.cpp)
target_sources(ZCodePet PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/resources.rc)
target_sources(ZCodePet PRIVATE src/composition_device.cpp)
file(GLOB SPINE_SOURCES CONFIGURE_DEPENDS third_party/spine-cpp/src/spine/*.cpp)
add_library(spine-cpp STATIC ${SPINE_SOURCES})
target_include_directories(spine-cpp PUBLIC third_party/spine-cpp/include)
target_compile_options(spine-cpp PRIVATE /utf-8 /EHsc /W0)
target_sources(ZCodePet PRIVATE src/spine_pet.cpp src/embedded_assets.cpp)
target_link_libraries(ZCodePet PRIVATE spine-cpp d3dcompiler windowscodecs)
target_compile_definitions(ZCodePet PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ZCodePet PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(ZCodePet PRIVATE d2d1 dwrite dwmapi gdiplus shell32 ole32 oleaut32 user32 gdi32 dxgi uiautomationcore d3d11 windowsapp CoreMessaging)
enable_testing()
add_test(NAME observer COMMAND ZCodePet --self-test)
add_test(NAME presentation COMMAND ZCodePet --presentation-test)

add_test(NAME badge COMMAND ZCodePet --badge-test)
add_test(NAME placement COMMAND ZCodePet --placement-test)
add_test(NAME floating-layout COMMAND ZCodePet --floating-test)
add_test(NAME popup-motion COMMAND ZCodePet --popup-motion-test)
add_test(NAME navigation COMMAND ZCodePet --navigation-test)
add_test(NAME interaction-motion COMMAND ZCodePet --interaction-motion-test)
add_test(NAME notices COMMAND ZCodePet --notice-test)
add_test(NAME panel-scene COMMAND ZCodePet --panel-scene-test)
add_test(NAME composition COMMAND ZCodePet --composition-test)
add_test(NAME raster COMMAND ZCodePet --raster-test)
add_test(NAME spine-model COMMAND ZCodePet --spine-model-test)
add_test(NAME frame-clock COMMAND ZCodePet --frame-clock-test)
add_test(NAME spine-lifecycle COMMAND ZCodePet --spine-lifecycle-test)
add_test(NAME notice-reflow COMMAND ZCodePet --notice-reflow-test)

add_test(NAME embedded-assets COMMAND ZCodePet --embedded-assets-test)
add_test(NAME standalone COMMAND ${CMAKE_COMMAND}
  "-DEXE=$<TARGET_FILE:ZCodePet>" "-DROOT=${CMAKE_BINARY_DIR}"
  -P "${CMAKE_SOURCE_DIR}/scripts/test-standalone.cmake")

add_library(zstd-decoder STATIC third_party/zstd/zstddeclib.c)
target_compile_options(zstd-decoder PRIVATE /W0)
target_link_libraries(ZCodePet PRIVATE zstd-decoder)
add_test(NAME data-sources COMMAND ZCodePet --data-sources-test)
