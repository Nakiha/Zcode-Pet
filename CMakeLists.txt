cmake_minimum_required(VERSION 3.24)
project(ZCodePet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(ZCodePet WIN32 src/main.cpp src/observer.cpp src/toolbar.cpp src/status_badge.cpp src/notice_close.cpp src/popup_motion.cpp src/frame_clock.cpp src/session_navigation.cpp src/interaction_motion.cpp)
target_compile_definitions(ZCodePet PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ZCodePet PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(ZCodePet PRIVATE d2d1 dwrite dwmapi gdiplus shell32 ole32 oleaut32 user32 gdi32 dxgi uiautomationcore)
add_custom_command(TARGET ZCodePet POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:ZCodePet>/assets")
enable_testing()
add_test(NAME observer COMMAND ZCodePet --self-test)
add_test(NAME presentation COMMAND ZCodePet --presentation-test)

add_test(NAME badge COMMAND ZCodePet --badge-test)
add_test(NAME placement COMMAND ZCodePet --placement-test)
add_test(NAME floating-layout COMMAND ZCodePet --floating-test)
add_test(NAME popup-motion COMMAND ZCodePet --popup-motion-test)
add_test(NAME navigation COMMAND ZCodePet --navigation-test)
add_test(NAME interaction-motion COMMAND ZCodePet --interaction-motion-test)
